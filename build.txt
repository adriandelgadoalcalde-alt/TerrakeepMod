# NoSync: es una herramienta local (interfaz + edicion del personaje/mundo de la partida
# propia). No añade contenido que haya que sincronizar entre cliente y servidor, asi que ni
# obliga al servidor a tenerlo ni exige que las dos partes coincidan.
# Ventaja practica añadida, y no menor: con NoSync el mod TAMBIEN carga en el servidor
# dedicado, que arranca sin ventana y sin audio. Eso permite verificar de forma automatica y
# reproducible que el .tmod carga y que lib/Terrakeep.Core.dll (compilado para net8) se
# resuelve y EJECUTA dentro del runtime .NET 8 de tModLoader, sin depender de que haya una
# sesion de escritorio interactiva. Ver bitacora.md, entrada de WS0.
side = NoSync

# El DLL de Terrakeep.Core (compilado para net8.0, el runtime real de tModLoader) vive
# en lib\Terrakeep.Core.dll. ModCompile.DllRefPath lo busca literalmente ahi
# (Path.Combine(mod.path, "lib", dllName) + ".dll") y ModCompile.PackageMod lo mete dentro
# del .tmod como "lib/Terrakeep.Core.dll".
dllReferences = Terrakeep.Core

# El .csproj, los .md de documentacion y las carpetas de trabajo no pintan nada dentro del
# .tmod. Los .cs ya los excluye tModLoader por su cuenta salvo includeSource = true.
#
# scripts\* y evidencia\* son el ARNES DE PRUEBAS de este repositorio (los .ps1 de verificacion
# y los logs reales de cada ejecucion). Estaban entrando dentro del .tmod publicado: se vio
# listando el archivo con "node tmod-extract.js" y salian los 12 scripts y los 12 logs, ~180 KB
# de cosas que a quien instale el mod no le sirven de nada. ModCompile.PackageMod empaqueta
# TODO lo que no descarte IgnoreResource, asi que la unica forma de dejarlos fuera es nombrarlos
# aqui.
#
# bin-*, obj-* y *.pdb añadidos el 14-sep-2026 (hallazgo real de KeepQA): ModCompile.
# IgnoreCompletely (tModLoader.dll real, decompilado con ilspycmd) solo descarta rutas que
# empiezan LITERALMENTE por "bin\"/"obj\" - las carpetas de salida ALTERNATIVAS que dejan las
# verificaciones al compilar con -p:BaseOutputPath aparte (bin-checkDebug\, obj-guia\..., ya
# en .gitignore, nunca parte del mod) no coinciden con ese prefijo exacto y se colaban enteras
# dentro del .tmod publicado (project.assets.json, *.nuget.cache, rutas absolutas de esta
# maquina...). El .pdb tampoco lo descarta ninguna regla: ModCompile.Build lo añade siempre que
# RoslynCompile lo genere (codigo real, sin ninguna opcion de "Release sin simbolos" - no es un
# ajuste de Configuration que se pueda tocar, tModLoader no expone ese concepto en su propio
# -build), asi que la unica forma real de dejarlo fuera del paquete distribuible es nombrarlo
# aqui. El *.pdb de esta lista es documentacion de intencion, no una exclusion real: ver
# scripts\limpiar-tmod.ps1 (paso aparte tras compilar, necesario de verdad).
#
# docs\* añadido el 15-sep-2026: capturas reales del README (docs\screenshots\), documentacion
# pura, cero uso en tiempo de ejecucion - mismo motivo que scripts\/evidencia\.
#
# TerrakeepMod.Tests*, TerrakeepMod.LogicaPura* y dotnet-tools.json añadidos el 17-sep-2026
# (KeepQA, paso 2/3 de la integracion de CsCheck+Stryker.NET - mismo hallazgo real y mismo patron
# que TModLoaderMod.Pruebas.Unit* en el build.txt del repo hermano TModLoaderMod, ver el suyo):
# el proyecto de tests xUnit+CsCheck y la biblioteca minima que enlaza GramaticaBusqueda.cs para
# Stryker.NET viven DENTRO de esta misma carpeta de mod. Sin este buildIgnore,
# ModCompile.CompileMod (decompilado: Directory.GetFiles(mod.path, "*.cs",
# SearchOption.AllDirectories), filtrado por IgnoreCompletely -> ignoreFile -> buildIgnore; NO
# pasa por MSBuild ni por el <Compile Remove> del .csproj, que solo protege "dotnet build"/el IDE)
# recogia tambien esos .cs (con "using Xunit"/"using CsCheck" que el mod no tiene) y los generados
# de sus propios bin/obj, y el -build real fallaba con mas de 100 errores CS0246/CS0579
# (confirmado en vivo, 17-sep-2026, antes de este buildIgnore). La regex de FitsMask
# (BuildProperties.cs) no ancla "*" a un segmento de ruta, asi que un solo "*" ya equivale a "**"
# (cruza barras) - "TerrakeepMod.Tests*"/"TerrakeepMod.LogicaPura*" bastan para toda la subcarpeta.
# dotnet-tools.json (manifiesto local de "dotnet tool" que fija dotnet-stryker 5.0.0) tampoco tiene
# extension .cs/.csproj/.sln, asi que sin nombrarlo aqui ModCompile.PackageMod lo habria
# empaquetado dentro del .tmod como recurso suelto - infraestructura de desarrollo, cero uso en
# tiempo de ejecucion para quien instale el mod.
buildIgnore = *.csproj, *.user, *.md, .git\*, .vs\*, .idea\*, obj\*, bin\*, bin-*, obj-*, *.pdb, Properties\*, scripts\*, evidencia\*, docs\*, TerrakeepMod.Tests*, TerrakeepMod.LogicaPura*, dotnet-tools.json
